---before setup:


[root@anuragrac1 /]# mkdir sybase


[root@anuragrac1 /]# adduser anurag


[root@anuragrac1 ~]# passwd anurag


[root@anuragrac1 /]# chown -R anurag /sybase


---after setup:


//go to installed folder and run below file


[root@anuragrac1 sybase]# ./setup.bin


// goto user directory


[root@anuragrac1 ~]# cd home/anurag


[root@anuragrac1 anurag]# ls -a


// goto install folder: sybase/ASE-15_0/install and run following command


[root@anuragrac1 sybase_inst]# cat /sybase_inst/SYBASE.sh >> /home/anurag/.bash_profile


// run ASE server file first


[anurag@anuragrac1 install]$  startserver -f RUN_sybase_new


// then run backup server file


[root@anuragrac1 install]# startserver -f RUN_sybase_new_BS


[root@anuragrac1 install]# showserver


[root@anuragrac1 install]# isql -Usa -Ssybase_new
