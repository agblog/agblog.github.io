📜 Script 1: 15-Minute Monitoring Block (dbcheck.sh)

#!/bin/bash
# File name : dbcheck.sh

REPORT=/var/opt/mssql/backup/monitoring/reportstemp/dbcheck.$$
SERVER=`hostname`

echo -n 'Report for host : ' > $REPORT
echo $SERVER >> $REPORT

# Gather System Metrics
echo 'Server Uptime' >> $REPORT; uptime >> $REPORT
echo 'Cluster Status' >> $REPORT; /usr/sbin/crm status >> $REPORT
echo 'DB Drive Space Details' >> $REPORT; df -h >> $REPORT
echo 'SQL Server Status' >> $REPORT; systemctl status mssql-server -q | head -5 >> $REPORT

# Query Live SQL Performance Metrics
/opt/mssql-tools/bin/sqlcmd -S localhost -U anurag_dba -P password@ -Q "exec querytool..cpuutilization" >> $REPORT
/opt/mssql-tools/bin/sqlcmd -S localhost -U anurag_dba -P password@ -Q "exec querytool..CheckPLE" >> $REPORT
/opt/mssql-tools/bin/sqlcmd -S localhost -U anurag_dba -P password@ -Q "exec querytool..activesessions2" >> $REPORT
/opt/mssql-tools/bin/sqlcmd -S localhost -U anurag_dba -P password@ -Q "exec querytool..agmonitor1" >> $REPORT
/opt/mssql-tools/bin/sqlcmd -S localhost -U anurag_dba -P password@ -Q "exec querytool..last1hrjobstatus" >> $REPORT

# Filter Errors Last 15 Min
awk -v d1="$(date --date="15 minutes ago" "+%Y-%m-%d %H:%M")" -v d2="$(date "+%Y-%m-%d %H:%M")" '$0 > d1 && $0 < d2 || $0 ~ d2' /var/opt/mssql/log/errorlog >> $REPORT

# Dispatch automated email via mailutils
cat $REPORT | mail -s "AnuragTest_$SERVER" anurag.dobade@gmail.com
rm $REPORT
