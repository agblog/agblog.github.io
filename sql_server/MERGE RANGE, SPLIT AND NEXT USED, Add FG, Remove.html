We have 1000 rows in each filegroup PRIMARY(2014), DATA2015, DATA2016


Adding records with 2017 -2018 values it so goes to data2016 filegroup

SET NOCOUNT ON;
    SET STATISTICS IO OFF;
   
    DECLARE @FromDate date = '2017-01-01';
    DECLARE @ToDate date =   '2018-01-01';
   
    INSERT INTO dbo.PartitionedTable
    SELECT 
        REPLICATE('A',10),
        REPLICATE('B',10),
        DATEADD(DD,FLOOR(RAND()*(DATEDIFF(DD,@FromDate,@ToDate))),@FromDate);
    GO 1000
